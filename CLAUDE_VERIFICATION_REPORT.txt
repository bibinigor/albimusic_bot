def add_balance(user_id, amount):
    """Добавление баланса пользователю (PostgreSQL версия)"""
    try:
        from db_utils import execute_query_sync
        execute_query_sync('UPDATE users SET balance = balance + %s WHERE user_id = %s', (amount, user_id))
        logging.info(f"✅ Баланс пользователя {user_id} увеличен на {amount}")
    except Exception as e:
        logging.error(f"❌ Ошибка добавления баланса {user_id}: {e}")

def save_user_token(user_id, access_token):
    """Сохранение токена пользователя (PostgreSQL версия)"""
    try:
        from db_utils import execute_query_sync
        execute_query_sync('INSERT INTO user_tokens (user_id, access_token, created_at) VALUES (%s, %s, NOW()) ON CONFLICT (user_id) DO UPDATE SET access_token = EXCLUDED.access_token, created_at = NOW()', (user_id, access_token))
        logging.info(f"✅ Токен пользователя {user_id} сохранен")
    except Exception as e:
        logging.error(f"❌ Ошибка сохранения токена {user_id}: {e}")

def add_channel_post(user_id, audio_url, comment, message_id=None):
    """Добавление записи о публикации в канал (PostgreSQL версия)"""
    try:
        from db_utils import execute_query_sync
        execute_query_sync('INSERT INTO channel_posts (user_id, audio_url, comment, message_id, posted_at) VALUES (%s, %s, %s, %s, NOW())', (user_id, audio_url, comment, message_id))
        logging.info(f"✅ Публикация сохранена для пользователя {user_id}")
        return True
    except Exception as e:
        logging.error(f"❌ Ошибка сохранения публикации {user_id}: {e}")
        return False

def get_user_token(user_id):
    """Получение токена пользователя (PostgreSQL версия)"""
    try:
        from db_utils import execute_query_sync
        result = execute_query_sync('SELECT access_token FROM user_tokens WHERE user_id = %s', (user_id,), fetch_one=True)
        if result:
            return result[0]
        return None
    except Exception as e:
        logging.error(f"❌ Ошибка получения токена {user_id}: {e}")
        return None
