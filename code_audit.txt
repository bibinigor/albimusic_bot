# Функция save_generation_task_sync (исправленная версия):

def save_generation_task_sync(user_id, task_id, prompt, status, audio_url=None):
    """Синхронное сохранение/обновление задачи в БД (UPSERT)"""
    logger = logging.getLogger(__name__)
    
    try:
        logger.info(f"💾 save_generation_task_sync: task_id={task_id}, status={status}, audio_url={'present' if audio_url else 'None'}")
        
        if audio_url is not None:
            # Если есть audio_url, обновляем существующую запись
            logger.info(f"📝 Обновление с audio_url: task_id={task_id}")
            if task_id is None:
                logger.error("❌ task_id is None! Невозможно обновить запись")
                return {"success": False, "error": "task_id is None"}
            result = execute_query_sync(
                """UPDATE generations 
                SET status = %s, 
                    audio_url = %s
                WHERE task_id = %s
                RETURNING id""",
                (status, audio_url, task_id)
            )
            
            if result:
                logger.info(f"✅ Задача {task_id} обновлена в БД (status: {status})")
                return {'success': True, 'id': result[0]}
            else:
                logger.warning(f"⚠️  Задача {task_id} не найдена при обновлении")
                return {'success': False, 'error': 'Задача не найдена'}
                
        else:
            # Если нет audio_url, пробуем обновить статус существующей записи
            logger.info(f"📝 Обновление статуса: task_id={task_id}, status={status}")
            if task_id is None:
                logger.error("❌ task_id is None! Невозможно обновить статус")
                return {"success": False, "error": "task_id is None"}
            result = execute_query_sync(
                """UPDATE generations 
                SET status = %s
                WHERE task_id = %s
                RETURNING id""",
                (status, task_id)
            )
            
            if result:
                logger.info(f"✅ Статус задачи {task_id} обновлен на '{status}'")
                return {'success': True, 'id': result[0]}
            else:
                # Если запись не найдена, создаем новую
                logger.info(f"📝 Создание новой записи: task_id={task_id}")
                result = execute_query_sync(
                    """INSERT INTO generations 
                    (user_id, task_id, prompt, status, created_at) 
                    VALUES (%s, %s, %s, %s, NOW())
                    RETURNING id""",
                    (user_id, task_id, prompt, status)
                )
                if result:
                    logger.info(f"✅ Новая задача создана: id={result[0]}")
                    return {'success': True, 'id': result[0]}
                else:
                    logger.error(f"❌ Ошибка создания новой задачи")
                    return {'success': False, 'error': 'Не удалось создать задачу'}
                    
    except Exception as e:
        logger.error(f"❌ Ошибка сохранения задачи {task_id}: {e}", exc_info=True)
        return {'success': False, 'error': f'Ошибка обновления задачи: {str(e)}'}
